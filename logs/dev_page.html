Chi Tiết Đề Bài

Cho đồ thị mạng luồng đơn hướng có sức chứa trên mỗi cung, đỉnh phát nguồn (Source) và đỉnh thu (Sink). Hãy tìm tổng luồng chảy lớn nhất có thể bơm từ đỉnh nguồn sang đỉnh thu. Thuật toán Dinic tối ưu hóa quá trình tăng luồng qua đồ thị phân tầng (Level Graph) và tìm luồng cản (Blocking Flow).

Khởi Tạo Thuật Toán Dinic

Bắt đầu thuật toán Dinic tìm Luồng cực đại từ S = 0 đến T = 5. Đồ thị gồm 6 đỉnh và 9 cạnh có hướng.

1#include <vector>
2#include <queue>
3#include <algorithm>
4
5using namespace std;
6
7struct Edge {
8    int to;
9    int cap;
10    int flow;
11    int rev;
12};
13
14class Dinic {
15    int n, s, t;
16    vector<vector<Edge>> adj;
17    vector<int> level, ptr;
18
19public:
20    Dinic(int n, int s, int t) : n(n), s(s), t(t), adj(n), level(n), ptr(n) {}
21
22    void addEdge(int from, int to, int cap) {
23        adj[from].push_back({to, cap, 0, (int)adj[to].size()});
24        adj[to].push_back({from, 0, 0, (int)adj[from].size() - 1});
25    }
26
27    bool bfs() {
28        fill(level.begin(), level.end(), -1);
29        level[s] = 0;
30        queue<int> q;
31        q.push(s);
32        while (!q.empty()) {
33            int u = q.front();
34            q.pop();
35            for (const auto& edge : adj[u]) {
36                if (edge.cap - edge.flow > 0 && level[edge.to] == -1) {
37                    level[edge.to] = level[u] + 1;
38                    q.push(edge.to);
39                }
40            }
41        }
42        return level[t] != -1;
43    }
44
45    int dfs(int u, int pushed) {
46        if (pushed == 0 || u == t) return pushed;
47        for (int& cid = ptr[u]; cid < (int)adj[u].size(); ++cid) {
48            auto& edge = adj[u][cid];
49            int trg = edge.to;
50            if (level[u] + 1 != level[trg] || edge.cap - edge.flow == 0) continue;
51            int tr = dfs(trg, min(pushed, edge.cap - edge.flow));
52            if (tr == 0) continue;
53            edge.flow += tr;
54            adj[trg][edge.rev].flow -= tr;
55            return tr;
56        }
57        return 0;
58    }
59
60    int maxFlow() {
61        int flow = 0;
62        while (bfs()) {
63            fill(ptr.begin(), ptr.end(), 0);
64            while (int pushed = dfs(s, 1e9)) {
65                flow += pushed;
66            }
67        }
68        return flow;
69    }
70};
